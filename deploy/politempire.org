# politempire.org — ОСНОВНОЙ домен сайта (Next.js) + authlib-прокси.
# Защита от DDoS: include snippets/antiddos-server.conf + per-location лимиты.
#
# Установка:
#   sudo cp deploy/nginx-politempire-org.conf /etc/nginx/sites-available/politempire.org
#   sudo ln -sf /etc/nginx/sites-available/politempire.org /etc/nginx/sites-enabled/politempire.org
#   sudo nginx -t && sudo systemctl reload nginx

# ---- HTTP → HTTPS -----------------------------------------------------------
server {
    listen 80;
    listen [::]:80;
    server_name politempire.org www.politempire.org;

    location /.well-known/acme-challenge/ {
        root /var/www/html;
    }

    # GML auth: отдаём напрямую по HTTP (без HTTPS-редиректа),
    # чтобы GML-бэкенд в Docker мог аутентифицировать без SSL-ошибок.
    location /api/gml/auth {
        proxy_pass http://127.0.0.1:3001;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 10s;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

# ---- Основной сайт ----------------------------------------------------------
server {
    listen 443 ssl default_server;
    listen [::]:443 ssl default_server;
    http2 on;
    server_name politempire.org www.politempire.org;

    ssl_certificate     /etc/letsencrypt/live/politempire.org-0001/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/politempire.org-0001/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    client_max_body_size 100M;

    # === Anti-DDoS: лимит соединений, блок bad UA/method/URI, общий rate-limit.
    include snippets/antiddos-server.conf;

    # Не пишем в access.log мусорные 429/400/444 — во время атаки они забивают
    # диск (было 9GB/день). $loggable определён в conf.d/00-antiddos.conf.
    # 403/404/405 и 2xx/3xx/5xx логируются как обычно — нужны для fail2ban
    # (nginx-forbidden, nginx-botsearch, nginx-post-root) и отладки.
    access_log /var/log/nginx/access.log combined if=$loggable;

    # ---- Authlib для игры (joinserver/hasJoined) — на GML-бэкенд (5003) -----
    # Высокий лимит: реальные Minecraft-клиенты делают много параллельных
    # запросов при входе/смене сервера. 30r/s + burst 60 — комфортно для игры,
    # но всё равно режет DDoS.
    location /api/v1/profiles/info {
        limit_req zone=antiddos_authlib burst=500 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        rewrite ^ /api/v1/profiles/details break;
        proxy_pass http://127.0.0.1:5004;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Authorization $fixed_auth;
        proxy_read_timeout 60s;
    }
    # Файлы сборки — напрямую на GML API (YARP, порт 5003), минуя gml-proxy.js.
    # Высокий burst (120): лаунчер скачивает сотни файлов подряд.
    location /api/v1/file/ {
        limit_req zone=antiddos_authlib burst=1500 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        proxy_pass http://127.0.0.1:5003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_buffering off;
        proxy_read_timeout 120s;
        add_header Cache-Control "public, max-age=604800, immutable";
    }
    # Файлы сборки по пути: nginx static + Range — докачка оборванных файлов
    # (см. комментарий в politempire.ru).
    location /files/ {
        alias /opt/polit-empire/data/GmlBackend/;
        limit_req zone=antiddos_authlib burst=1500 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        gzip off;
        sendfile on;
        tcp_nopush on;
        send_timeout 300s;
        add_header Accept-Ranges bytes always;
        add_header Cache-Control "public, max-age=604800, immutable" always;
    }
    location /api/v1/ {
        limit_req zone=antiddos_authlib burst=500 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        limit_req zone=antiddos_backend_org burst=100 nodelay;
        proxy_pass http://127.0.0.1:5004;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 60s;
    }

    # ---- API плагина Minecraft-сервера (PoliteShopMod) ----------------------
    # Плагин на сервере делает много параллельных запросов от имени всех
    # игроков: /api/mod/balance, /api/mod/admin/dc, /api/mod/claim и т.п.
    # Поэтому отдельная зона antiddos_mod с высоким лимитом 20r/s + burst 80.
    # IP самого Minecraft-сервера в whitelist (geo $is_trusted_ip в
    # conf.d/00-antiddos.conf) — для него лимита нет вообще.
    location /api/mod/ {
        limit_req zone=antiddos_mod burst=300 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
    }

    # ---- Телеметрия лаунчера (logs/heartbeat/hwid/stats) ---------------------
    # Лаунчер шлёт логи пачками каждые ~2с плюс heartbeat — на antiddos_api
    # (2r/s, burst 5) это ловит 429 у обычного играющего человека. Отдельная
    # зона с высоким лимитом; global-лимит оставляем как потолок на всех.
    location ~ ^/api/(launcher/(logs|heartbeat|hwid)|player/stats/sync)$ {
        limit_req zone=antiddos_authlib burst=500 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 60s;
    }

    # ---- API сайта (/api/auth, /api/telegram, /api/launcher, /api/telemetry)
    # 20r/s + burst 500 на игрока: лаунчер при старте делает пакет запросов
    # (news, version, heartbeat, logs, verify, stats), скачивание файлов
    # идёт по /files/ и /api/v1/file/ (зона authlib), не сюда.
    location /api/ {
        limit_req zone=antiddos_api burst=500 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
    }

    # ---- Корень сайта — только GET/HEAD (блокируем POST / — основная атака) -
    # `limit_except GET HEAD OPTIONS` отдаёт 403 на любые другие методы (POST,
    # PUT, TRACE, CONNECT). Легитимный сайт никогда не получает POST на корень.
    location = / {
        limit_except GET HEAD OPTIONS {
            deny all;
        }
        limit_req zone=antiddos_general burst=150 nodelay;
        limit_req zone=antiddos_auth burst=10 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
        add_header Cache-Control "public, s-maxage=120" always;
    }

    # ---- Статика Next.js — прокси на контейнер (standalone-сборка) ------------
    location /_next/static/ {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        add_header Cache-Control "public, immutable, max-age=31536000";
    }
    location /favicon.ico {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        expires 7d;
    }

    # ---- Всё остальное (страницы, статика) — прокси на Next.js ---------------
    location / {
        limit_req zone=antiddos_general burst=150 nodelay;
        limit_req zone=antiddos_auth burst=10 nodelay;
        limit_req zone=antiddos_global_org burst=300 nodelay;
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
        add_header Cache-Control "public, s-maxage=120" always;
    }
}
